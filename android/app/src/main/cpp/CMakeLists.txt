cmake_minimum_required(VERSION 3.22.1)
project(readage_hymt2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(READAGE_ARM_CPU_VARIANTS "Select a compatible optimized ARM CPU backend at runtime" ON)
option(READAGE_BUILD_TRANSLATION_BENCH "Build the isolated real-model benchmark runner" OFF)
option(READAGE_VULKAN "Build experimental ARM64 Vulkan support (not the production default)" OFF)
if(ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a" AND READAGE_ARM_CPU_VARIANTS)
    set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
else()
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
endif()
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMIT "8ae386707bed05f02354dbfc6fa5ee0cf58e5f0f" CACHE STRING "" FORCE)
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp-8ae386707bed05f02354dbfc6fa5ee0cf58e5f0f")
if(READAGE_VULKAN AND ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a")
    if(NOT READAGE_ARM_CPU_VARIANTS)
        message(FATAL_ERROR "Experimental Vulkan requires packaged CPU variants for fallback.")
    endif()
    include(cmake/ReadageVulkan.cmake)
else()
    # Switching an existing build back to CPU must also clear ggml's cached ON flag.
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
endif()
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Missing pinned llama.cpp source. Run node scripts/prepare-android-translation.mjs --download before building.")
endif()
add_subdirectory("${LLAMA_SOURCE}" llama)
if(TARGET ggml-vulkan)
    target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
endif()
add_library(readage_translation_core STATIC translation_core.cpp)
target_link_libraries(readage_translation_core PUBLIC llama)
target_include_directories(readage_translation_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(TARGET ggml-vulkan)
    target_compile_definitions(readage_translation_core PRIVATE READAGE_VULKAN)
endif()
if(ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a" AND READAGE_ARM_CPU_VARIANTS)
    target_compile_definitions(readage_translation_core PRIVATE READAGE_ARM_CPU_VARIANTS)
endif()
if(ANDROID)
    add_library(readage_hymt2 SHARED translation_jni.cpp)
    target_link_libraries(readage_hymt2 PRIVATE readage_translation_core log)
    # Android 15+ supports 16 KiB pages. NDK 28 already defaults to this alignment.
    target_link_options(readage_hymt2 PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(readage_translation_host translation_host.cpp)
    target_link_libraries(readage_translation_host PRIVATE readage_translation_core)
endif()
if(READAGE_BUILD_TRANSLATION_BENCH)
    add_executable(readage_translation_bench translation_bench.cpp)
    target_link_libraries(readage_translation_bench PRIVATE readage_translation_core)
endif()
